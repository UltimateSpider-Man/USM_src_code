cmake_minimum_required(VERSION 3.16)
project(usm_online_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(online_protocol test_protocol.cpp)
add_executable(online_session test_session.cpp ../../src/multiplayer_online_session.cpp)
foreach(t online_protocol online_session)
 target_include_directories(${t} PRIVATE ../../src)
 if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-misleading-indentation)
 endif()
endforeach()
if(WIN32)
 target_link_libraries(online_session PRIVATE ws2_32 bcrypt)
endif()
add_test(NAME online_wire_interpolation_validation COMMAND online_protocol)
add_test(NAME online_eight_player_tcp_udp COMMAND online_session)
set_tests_properties(online_eight_player_tcp_udp PROPERTIES TIMEOUT 25)

add_executable(online_ui test_ui.cpp)
add_executable(online_mode test_mode.cpp ../../src/multiplayer_online_session.cpp)
foreach(t online_ui online_mode)
 target_include_directories(${t} PRIVATE ../../src .)
 if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-misleading-indentation)
 endif()
endforeach()
if(WIN32)
 target_link_libraries(online_mode PRIVATE ws2_32 bcrypt)
endif()
add_test(NAME online_editor_analog_gamepad COMMAND online_ui)
add_test(NAME online_production_menu_lifecycle COMMAND online_mode)

add_executable(online_probe online_probe.cpp ../../src/multiplayer_online_session.cpp ../../src/multiplayer_online_interfaces.cpp)
target_include_directories(online_probe PRIVATE ../../src)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(online_probe PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-misleading-indentation)
endif()
if(WIN32)
 target_link_libraries(online_probe PRIVATE ws2_32 bcrypt iphlpapi)
endif()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
 add_test(NAME online_separate_processes_and_invalid_traffic
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_processes.py $<TARGET_FILE:online_probe>)
 set_tests_properties(online_separate_processes_and_invalid_traffic PROPERTIES TIMEOUT 15)
endif()
